cmake_minimum_required(VERSION 3.18)
project(cuda_cluster_transport LANGUAGES CXX)

option(USE_CUDA "Compile native CUDA GPU backend when nvcc is available" ON)
option(TRANSPORT_SINGLE_PRECISION "Use float instead of double for Real" OFF)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

include(CheckLanguage)
set(HAVE_CUDA_COMPILER FALSE)
if(USE_CUDA)
    check_language(CUDA)
    if(CMAKE_CUDA_COMPILER)
        enable_language(CUDA)
        set(HAVE_CUDA_COMPILER TRUE)
        set(CMAKE_CUDA_STANDARD 17)
        set(CMAKE_CUDA_STANDARD_REQUIRED ON)
        if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
            set(CMAKE_CUDA_ARCHITECTURES 70 75 80 86 89 90)
        endif()
        message(STATUS "CUDA compiler detected: ${CMAKE_CUDA_COMPILER} (building native GPU kernels)")
    else()
        message(STATUS "No CUDA compiler (nvcc) found; building portable C++17 host backend")
    endif()
else()
    message(STATUS "USE_CUDA=OFF; building portable C++17 host backend")
endif()

function(configure_transport_target target_name src_file)
    if(HAVE_CUDA_COMPILER)
        add_executable(${target_name} ${src_file})
        set_source_files_properties(${src_file} PROPERTIES LANGUAGE CUDA)
        target_compile_options(${target_name} PRIVATE
            $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr -O3>
        )
    else()
        set_source_files_properties(${src_file} PROPERTIES LANGUAGE CXX)
        add_executable(${target_name} ${src_file})
        target_compile_options(${target_name} PRIVATE -x c++ -O3 -Wall -Wextra)
    endif()

    target_include_directories(${target_name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
    if(TRANSPORT_SINGLE_PRECISION)
        target_compile_definitions(${target_name} PRIVATE TRANSPORT_SINGLE_PRECISION=1)
    endif()
endfunction()

configure_transport_target(cluster_transport src/main.cu)
configure_transport_target(cluster_transport_tests src/tests.cu)

configure_transport_target(cluster_transport_fp32 src/main.cu)
target_compile_definitions(cluster_transport_fp32 PRIVATE TRANSPORT_SINGLE_PRECISION=1)

enable_testing()
add_test(NAME physics_and_kernel_suite COMMAND cluster_transport_tests)
